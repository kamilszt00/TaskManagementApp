\section*{Użyte technologie}

\textbf{Backend}
\begin{itemize}
  \item Java 21
  \item Spring Boot 4 (Web, Data JPA, Security, Validation)
  \item JWT (biblioteka jjwt) -- uwierzytelnianie
  \item MySQL
  \item Lombok, MapStruct
  \item Maven
\end{itemize}

\textbf{Frontend}
\begin{itemize}
  \item Angular 22, TypeScript, RxJS
  \item Angular Material -- responsywny interfejs, dostosowany do urządzeń mobilnych
  \item Angular Service Worker (PWA)
\end{itemize}

\textbf{Narzędzia}
\begin{itemize}
  \item Docker
  \item Vitest, JUnit, AssertJ -- testy
\end{itemize}
